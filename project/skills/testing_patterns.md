# Testing Patterns — Bronze (pytest), Silver + Gold (SQL tests)

Complete reference for testing all three layers of the Securities Master Data pipeline.

---

## 1. Test Strategy by Layer

| Layer | Test Framework | What to Test | Generated By |
|-------|---------------|-------------|--------------|
| Bronze | pytest + PySpark | CSV read, metadata cols, MERGE behavior, schema drift | QA Agent |
| Silver | SQL assertions | DQ rule outcomes, SCD2 correctness, reject routing | QA Agent |
| Gold | SQL assertions | Mart grain, join completeness, Genie comment presence | QA Agent |
| Spec YAML | sml validate | Naming conventions vs CLAUDE.md | validate_spec_agent |

### Test: Metadata Columns Added Correctly

```python
# tests/bronze/test_bronze_loader.py
from src.ingestion.bronze_loader import _add_metadata

def test_metadata_columns_added(spark, sample_product_df):
    """All 5 metadata columns must be present after _add_metadata()."""
    result = _add_metadata(sample_product_df, "product.csv", "volume", "batch_001")

    expected_meta_cols = {"_source_file", "_source_type", "_ingestion_ts", "_batch_id", "_row_hash"}
    actual_cols = set(result.columns)

    assert expected_meta_cols.issubset(actual_cols), \
        f"Missing metadata columns: {expected_meta_cols - actual_cols}"

def test_row_hash_is_deterministic(spark, sample_product_df):
    """Same input data must produce the same _row_hash (required for MERGE INTO CDC)."""
    df1 = _add_metadata(sample_product_df, "product.csv", "volume", "batch_001")
    df2 = _add_metadata(sample_product_df, "product.csv", "volume", "batch_002")

    hashes1 = {row["product_id"]: row["_row_hash"] for row in df1.collect()}
    hashes2 = {row["product_id"]: row["_row_hash"] for row in df2.collect()}

    assert hashes1 == hashes2, "_row_hash changed between runs for identical data"

def test_source_file_recorded(spark, sample_product_df):
    """_source_file must match the provided source label."""
    result = _add_metadata(sample_product_df, "product.csv", "volume", "batch_001")
    source_files = {row["_source_file"] for row in result.collect()}
    assert source_files == {"product.csv"}

def test_row_count_preserved(spark, sample_product_df):
    """Metadata columns must not filter or duplicate rows."""
    result = _add_metadata(sample_product_df, "product.csv", "volume", "batch_001")
    assert result.count() == sample_product_df.count()
```

### Test: Schema Drift Detection

```python
# tests/bronze/test_schema_drift.py
from src.ingestion.schema_drift import detect_drift

def test_detect_new_table(spark, sample_product_df):
    """Non-existent table should return new_table=True."""
    result = detect_drift(spark, "nonexistent.catalog.table", sample_product_df.schema)
    assert result["new_table"] is True
    assert result["additive"] == []
    assert result["breaking"] == []

def test_detect_additive_drift(spark, tmp_path, sample_product_df):
    """New column in incoming data = additive drift."""
    from pyspark.sql import functions as F

    # Write existing table
    table_path = str(tmp_path / "product_table")
    sample_product_df.write.format("delta").save(table_path)
    spark.sql(f"CREATE TABLE IF NOT EXISTS test_product USING DELTA LOCATION '{table_path}'")

    # Incoming data has an extra column
    incoming = sample_product_df.withColumn("new_field", F.lit("new_value"))
    result = detect_drift(spark, "test_product", incoming.schema)

    assert "new_field" in result["additive"]
    assert result["breaking"] == []

def test_detect_breaking_drift(spark, tmp_path, sample_product_df):
    """Type change in existing column = breaking drift."""
    from pyspark.sql import functions as F
    from pyspark.sql.types import IntegerType

    table_path = str(tmp_path / "product_break")
    sample_product_df.write.format("delta").save(table_path)
    spark.sql(f"CREATE TABLE IF NOT EXISTS test_product_break USING DELTA LOCATION '{table_path}'")

    # Cast current_face_value to Integer (type change: DoubleType → IntegerType)
    incoming = sample_product_df.withColumn("current_face_value", F.col("current_face_value").cast(IntegerType()))
    result = detect_drift(spark, "test_product_break", incoming.schema)

    assert "current_face_value" in result["breaking"]
```

import pytest

## 3. Silver Tests (SQL Assertions)

Run these as Databricks SQL notebook cells or as pytest + Spark SQL:

```sql
-- tests/silver/test_rule0001_id_type.sql
-- RULE0001: No invalid id_type values in Silver
SELECT COUNT(*) AS violations
FROM statestreet.s_statestreet.product
WHERE is_current = TRUE
  AND id_type NOT IN ('CUSIP','ISIN','SEDOL','TICKER','BLOOMBERG_ID');
-- Expected: 0
```

```sql
-- tests/silver/test_scd2_integrity.sql
-- SCD2: each product_id must have exactly one current row
SELECT product_id, COUNT(*) AS current_versions
FROM statestreet.s_statestreet.product
WHERE is_current = TRUE
GROUP BY product_id
HAVING COUNT(*) > 1;
-- Expected: 0 rows (no duplicates)
```

```sql
-- tests/silver/test_scd2_sentinel_dates.sql
-- All current rows must have effective_end_date = 9999-12-31
SELECT COUNT(*) AS invalid_sentinels
FROM statestreet.s_statestreet.product
WHERE is_current = TRUE
  AND effective_end_date != DATE '9999-12-31';
-- Expected: 0
```

```sql
-- tests/silver/test_rejects_captured.sql
-- All reject tables must exist and be readable
SELECT 'product_rejects' AS table_name, COUNT(*) AS reject_count
FROM statestreet.s_statestreet.product_rejects
UNION ALL
SELECT 'bond_rejects',    COUNT(*) FROM statestreet.s_statestreet.bond_rejects
UNION ALL
SELECT 'legal_entity_rejects', COUNT(*) FROM statestreet.s_statestreet.legal_entity_rejects;
-- Expected: no error (tables exist)
```

```sql
-- tests/silver/test_referential_integrity.sql
-- Every bond in Silver must have a product record in Silver
SELECT COUNT(*) AS orphaned_bonds
FROM statestreet.s_statestreet.bond b
LEFT JOIN statestreet.s_statestreet.product p
  ON b.product_id = p.product_id AND p.is_current = TRUE
WHERE b.is_current = TRUE
  AND p.product_id IS NULL;
-- Expected: 0
```

```sql
-- tests/gold/test_dim_product_grain.sql
-- dim_product: exactly one row per product (no duplicates)
SELECT COUNT(*) AS total_products, COUNT(DISTINCT product_id) AS unique_products
FROM statestreet.g_statestreet.dim_product;
-- Expected: total_products == unique_products
```

```sql
-- tests/gold/test_dim_product_row_count.sql
-- dim_product should have all 200 source products
SELECT COUNT(*) AS product_count
FROM statestreet.g_statestreet.dim_product;
-- Expected: 200 (or >= expected count from data load)
```

```sql
-- tests/gold/test_fact_coupon_grain.sql
-- fact_coupon_schedule: one row per bond per payment date (no duplicates)
SELECT product_id, payment_date, COUNT(*) AS cnt
FROM statestreet.g_statestreet.fact_coupon_schedule
GROUP BY product_id, payment_date
HAVING COUNT(*) > 1;
-- Expected: 0 rows
```

```sql
-- tests/gold/test_genie_comments.sql
-- All Gold tables must have table-level comments (for Genie)
SELECT table_name, comment
FROM system.information_schema.tables
WHERE table_catalog = 'statestreet'
  AND table_schema   = 'g_statestreet'
  AND comment IS NULL;
-- Expected: 0 rows (all tables have comments)
```

```sql
-- tests/gold/test_dim_legal_entity_coverage.sql
-- All issuers referenced in dim_product must exist in dim_legal_entity
SELECT COUNT(*) AS missing_legal_entities
FROM statestreet.g_statestreet.dim_product p
LEFT JOIN statestreet.g_statestreet.dim_legal_entity le
  ON p.issuer_legal_entity_id = le.legal_entity_id
WHERE p.issuer_legal_entity_id IS NOT NULL
  AND le.legal_entity_id IS NULL;
-- Expected: 0
```

## 5. pytest Wrapper for SQL Tests

```python
# tests/silver/test_silver_layer.py
"""Pytest wrapper that runs SQL assertion tests against Databricks."""
import pytest

SILVER_TESTS = [
    ("RULE0001 id_type enum",
     """SELECT COUNT(*) AS cnt FROM statestreet.s_statestreet.product
        WHERE is_current = TRUE AND id_type NOT IN ('CUSIP','ISIN','SEDOL','TICKER','BLOOMBERG_ID')""",
     0),
    ("SCD2 no duplicate current rows",
     """SELECT COUNT(*) AS cnt FROM (
        SELECT product_id FROM statestreet.s_statestreet.product
        WHERE is_current = TRUE GROUP BY product_id HAVING COUNT(*) > 1)""",
     0),
    ("SCD2 sentinel date correct",
     """SELECT COUNT(*) AS cnt FROM statestreet.s_statestreet.product
        WHERE is_current = TRUE AND effective_end_date != DATE '9999-12-31'""",
     0),
]

@pytest.mark.parametrize("test_name,sql,expected", SILVER_TESTS)
def test_silver_assertion(spark, test_name, sql, expected):
    result = spark.sql(sql).first()["cnt"]
    assert result == expected, f"[FAIL] {test_name}: got {result}, expected {expected}"
```
